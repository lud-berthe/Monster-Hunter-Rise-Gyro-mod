cmake_minimum_required(VERSION 3.24)
project(MHRGyro VERSION 1.1.0 LANGUAGES C CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "MHRGyro requires a Windows x64 toolchain.")
endif()
include(CTest)
set(GYROLIB_SDK_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../../GyroLib/dist/sdk" CACHE PATH "Installed GyroLib 1.2 SDK (read-only)")
get_filename_component(GYROLIB_SDK_DIR "${GYROLIB_SDK_DIR}" ABSOLUTE BASE_DIR "${CMAKE_CURRENT_SOURCE_DIR}")
set(REFRAMEWORK_SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/third_party/REFramework" CACHE PATH "Pinned official REFramework source")
option(MHR_WITH_SDL "Include the GyroLib SDL reader" ON)
if(NOT EXISTS "${REFRAMEWORK_SOURCE_DIR}/include/reframework/API.h")
  message(FATAL_ERROR "Run tools/bootstrap.ps1 to fetch pinned REFramework SDK + Lua sources")
endif()
set(GyroLib_DIR "${GYROLIB_SDK_DIR}/lib/cmake/GyroLib")
if(NOT EXISTS "${GyroLib_DIR}/GyroLibConfig.cmake")
  message(FATAL_ERROR
    "Installed GyroLib SDK not found at ${GYROLIB_SDK_DIR}. "
    "Set -DGYROLIB_SDK_DIR=<installed-sdk-prefix> to a GyroLib 1.2 SDK. "
    "This project consumes the SDK without building or modifying GyroLib.")
endif()
# The bridge calls only GyroLib's C acquisition API; the bundled runtime owns SDL.
find_package(GyroLib 1.2 CONFIG REQUIRED COMPONENTS Core Overlay Steam)
get_target_property(_gyrolib_type GyroLib::gyrolib TYPE)
if(MHR_WITH_SDL)
  get_target_property(_gyrolib_definitions GyroLib::gyrolib INTERFACE_COMPILE_DEFINITIONS)
  if(NOT GyroLib_SDL_FOUND OR NOT "GL_BUNDLED_RUNTIME" IN_LIST _gyrolib_definitions
      OR NOT _gyrolib_type STREQUAL "SHARED_LIBRARY")
    message(FATAL_ERROR
      "MHR_WITH_SDL requires the Windows shared GyroLib SDK with its bundled SDL runtime. "
      "Select that SDK with GYROLIB_SDK_DIR, or use -DMHR_WITH_SDL=OFF for tests without hardware.")
  endif()
endif()
file(GLOB LUA_SOURCES CONFIGURE_DEPENDS "${REFRAMEWORK_SOURCE_DIR}/dependencies/lua/src/*.c")
list(FILTER LUA_SOURCES EXCLUDE REGEX "/(lua|luac)\\.c$")
add_library(mhr_lua STATIC ${LUA_SOURCES})
target_include_directories(mhr_lua PUBLIC "${REFRAMEWORK_SOURCE_DIR}/dependencies/lua/src")
add_library(mhr_bridge STATIC src/bridge.cpp src/window_reader.cpp src/native_gui.cpp)
target_compile_features(mhr_bridge PUBLIC cxx_std_17)
target_link_libraries(mhr_bridge PUBLIC GyroLib::gyrolib GyroLib::gyrolib_steam mhr_lua)
if(WIN32)
  target_link_libraries(mhr_bridge PUBLIC user32)
endif()
if(MHR_WITH_SDL)
  target_compile_definitions(mhr_bridge PRIVATE MHR_WITH_SDL=1)
endif()
if(WIN32)
  add_library(MHRGyro SHARED src/plugin.cpp)
  target_include_directories(MHRGyro PRIVATE "${REFRAMEWORK_SOURCE_DIR}/include")
  target_link_libraries(MHRGyro PRIVATE mhr_bridge)
  set_target_properties(MHRGyro PROPERTIES PREFIX "")
endif()
if(BUILD_TESTING)
  find_package(Python3 3.10 QUIET COMPONENTS Interpreter)
  if(Python3_Interpreter_FOUND)
    add_test(NAME mhr_distribution COMMAND "${Python3_EXECUTABLE}" -B "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_packaging.py")
    set_tests_properties(mhr_distribution PROPERTIES ENVIRONMENT "MHR_TEST_RUNNER=$<TARGET_FILE:mhr_tests>")
  endif()
  add_executable(mhr_recommended_tests tests/test_recommended.cpp)
  target_include_directories(mhr_recommended_tests PRIVATE src)
  target_link_libraries(mhr_recommended_tests PRIVATE mhr_bridge)
  add_test(NAME mhr_recommended_settings COMMAND mhr_recommended_tests)
  add_executable(mhr_steam_tests tests/test_steam_session.cpp)
  target_include_directories(mhr_steam_tests PRIVATE src)
  target_compile_features(mhr_steam_tests PRIVATE cxx_std_17)
  add_test(NAME mhr_steam_session COMMAND mhr_steam_tests)
  add_executable(mhr_bridge_benchmark tests/benchmark_bridge.cpp)
  target_include_directories(mhr_bridge_benchmark PRIVATE src)
  target_link_libraries(mhr_bridge_benchmark PRIVATE mhr_bridge)
  add_executable(mhr_tests tests/test_bridge.cpp)
  target_include_directories(mhr_tests PRIVATE src)
  target_link_libraries(mhr_tests PRIVATE mhr_bridge)
  add_test(NAME mhr_bridge_contract COMMAND mhr_tests "${CMAKE_CURRENT_SOURCE_DIR}")
  add_executable(mhr_flick_tests tests/test_flick_angles.cpp)
  target_link_libraries(mhr_flick_tests PRIVATE GyroLib::gyrolib)
  add_test(NAME mhr_sdk_flick_angles COMMAND mhr_flick_tests)
  add_executable(mhr_long_press_tests tests/test_long_press_filter.cpp)
  target_include_directories(mhr_long_press_tests PRIVATE src)
  target_compile_features(mhr_long_press_tests PRIVATE cxx_std_17)
  target_link_libraries(mhr_long_press_tests PRIVATE GyroLib::gyrolib)
  add_test(NAME mhr_sdk_long_press_filter COMMAND mhr_long_press_tests)
  if(WIN32)
    # Compile the dispatcher fixture directly so test synchronization hooks
    # never enter the installed plugin's bridge object.
    add_executable(mhr_window_tests tests/test_window_reader.cpp src/window_reader.cpp)
    target_compile_definitions(mhr_window_tests PRIVATE MHR_WINDOW_READER_TEST)
    target_include_directories(mhr_window_tests PRIVATE src)
    target_link_libraries(mhr_window_tests PRIVATE mhr_bridge)
    add_test(NAME mhr_window_transport COMMAND mhr_window_tests)
    add_executable(mhr_gui_tests tests/test_gui.cpp)
    target_include_directories(mhr_gui_tests PRIVATE src)
    target_link_libraries(mhr_gui_tests PRIVATE mhr_bridge d3d12 dxgi)
    add_test(NAME mhr_native_gui COMMAND mhr_gui_tests)
  endif()
endif()
if(WIN32 AND _gyrolib_type STREQUAL "SHARED_LIBRARY")
  # A DLL-only SDK update must refresh test/staging runtime even when no source
  # or import library changed and the linker has no reason to run again.
  add_custom_target(mhr_sdk_runtime ALL
    COMMAND ${CMAKE_COMMAND} -E make_directory "$<TARGET_FILE_DIR:MHRGyro>"
    COMMAND ${CMAKE_COMMAND} -E copy_if_different "$<TARGET_FILE:GyroLib::gyrolib>" "$<TARGET_FILE_DIR:MHRGyro>"
    VERBATIM)
  foreach(target MHRGyro mhr_bridge_benchmark mhr_tests mhr_recommended_tests mhr_flick_tests mhr_long_press_tests mhr_window_tests mhr_gui_tests)
    if(TARGET ${target})
      add_dependencies(${target} mhr_sdk_runtime)
    endif()
  endforeach()
  # Resolve paths through the imported target, including multi-config generators.
  # The staging script uses these exact build inputs to keep runtime/notices aligned.
  file(GENERATE OUTPUT "${CMAKE_BINARY_DIR}/mhr-stage-$<CONFIG>.txt"
    CONTENT "$<TARGET_FILE:MHRGyro>\n$<TARGET_FILE:GyroLib::gyrolib>\n${GYROLIB_SDK_DIR}/share/doc/GyroLib\n${REFRAMEWORK_SOURCE_DIR}\n")
endif()
